$msix = "$env:USERPROFILE\Desktop\看影音-2.1.103.msix"
Add-Type -AssemblyName System.IO.Compression
Add-Type -AssemblyName System.IO.Compression.FileSystem
Add-Type -AssemblyName System.Xml.Linq
$archive = [System.IO.Compression.ZipFile]::OpenRead($msix)
try {
  $manifestEntry = $archive.Entries |
    Where-Object { $_.FullName -eq 'AppxManifest.xml' } |
    Select-Object -First 1
  if ($null -eq $manifestEntry) { throw 'MSIX 中缺少 AppxManifest.xml' }

  }

  $archive.Dispose()

$identity = $manifest.Root.Elements() |
  Where-Object { $_.Name.LocalName -eq 'Identity' } |
  Select-Object -First 1
if ($null -eq $identity) { throw 'MSIX 清单缺少 Identity' }
$actualIdentity = [pscustomobject]@{

  Version = $identity.Attribute('Version').Value
  Architecture = $identity.Attribute('ProcessorArchitecture').Value
  Publisher = $identity.Attribute('Publisher').Value

$actualIdentity
if ($actualIdentity.Name -ne 'com.kanyingyin.player' -or

- [ ] **Step 3: 验证异机安装包内容和哈希**

```powershell
$zipPath = "$env:USERPROFILE\Desktop\看影音-2.1.103-异机安装包.zip"
Add-Type -AssemblyName System.IO.Compression
Add-Type -AssemblyName System.IO.Compression.FileSystem
$expectedNames = @(
  '看影音-2.1.103.msix',
  '看影音.cer',
  '安装看影音.ps1',
  '安装看影音.cmd',
  '安装说明.txt',
  'SHA256.txt'
) | Sort-Object
$archive = [System.IO.Compression.ZipFile]::OpenRead($zipPath)
try {
  $fileEntries = @($archive.Entries | Where-Object {
      -not $_.FullName.EndsWith('/')
    })
  $actualNames = @($fileEntries.FullName | Sort-Object)
  if (Compare-Object -ReferenceObject $expectedNames -DifferenceObject $actualNames) {
    throw '异机安装包文件清单不符合预期'
  }

  $msixEntry = $fileEntries |
    Where-Object { $_.FullName -eq '看影音-2.1.103.msix' } |
    Select-Object -First 1
  $msixStream = $msixEntry.Open()
  $sha256 = [System.Security.Cryptography.SHA256]::Create()
  try {
    $hashBytes = $sha256.ComputeHash($msixStream)
    $zipMsixHash = ([BitConverter]::ToString($hashBytes)).Replace('-', '')
  } finally {
    $sha256.Dispose()
    $msixStream.Dispose()
  }

  $desktopMsixHash = (Get-FileHash -LiteralPath "$env:USERPROFILE\Desktop\看影音-2.1.103.msix" -Algorithm SHA256).Hash
  if ($zipMsixHash -ne $desktopMsixHash) {
    throw 'ZIP 内 MSIX 与桌面 MSIX 哈希不一致'
  }

  $hashEntry = $fileEntries |
    Where-Object { $_.FullName -eq 'SHA256.txt' } |
    Select-Object -First 1
  $hashStream = $hashEntry.Open()
  $reader = [System.IO.StreamReader]::new($hashStream)
  try {
    $hashText = $reader.ReadToEnd()
  } finally {
    $reader.Dispose()
    $hashStream.Dispose()
  }
  if ($hashText -notmatch [regex]::Escape($desktopMsixHash)) {
    throw 'SHA256.txt 未记录正确的 MSIX 哈希'
  }
} finally {
  $archive.Dispose()
}

Get-FileHash -Algorithm SHA256 `
  "$env:USERPROFILE\Desktop\看影音-2.1.103.msix", $zipPath
```

Expected: ZIP 只包含固定六项，ZIP 内 MSIX 与桌面 MSIX 一致，`SHA256.txt` 记录正确哈希，并输出 MSIX 与 ZIP 的 SHA-256。
